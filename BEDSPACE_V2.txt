PROJECT: BEDSPACE_V2

PART 1 - DATABASE AND TABLES

CREATE DATABASE bedspace_v2;
USE bedspace_v2;

CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE,
    phone VARCHAR(15),
    gender ENUM('Male','Female','Other'),
    city VARCHAR(50)
);

CREATE TABLE hostels (
    hostel_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    type ENUM('PG','Hostel','Co-Living'),
    city VARCHAR(50) NOT NULL,
    address VARCHAR(255),
    rating DECIMAL(2,1),
    total_rooms INT
);

CREATE TABLE amenities (
    amenity_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) UNIQUE NOT NULL,
    category VARCHAR(50),
    icon VARCHAR(50)
);

CREATE TABLE hostel_amenities (
    id INT AUTO_INCREMENT PRIMARY KEY,
    hostel_id INT,
    amenity_id INT,
    is_free BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (hostel_id) REFERENCES hostels(hostel_id),
    FOREIGN KEY (amenity_id) REFERENCES amenities(amenity_id)
);

CREATE TABLE rooms (
    room_id INT AUTO_INCREMENT PRIMARY KEY,
    hostel_id INT,
    room_type ENUM('1-Sharing','2-Sharing','3-Sharing','4-Sharing'),
    rent_per_bed DECIMAL(8,2),
    capacity INT,
    floor INT,
    FOREIGN KEY (hostel_id) REFERENCES hostels(hostel_id)
);

CREATE TABLE beds (
    bed_id INT AUTO_INCREMENT PRIMARY KEY,
    room_id INT,
    bed_number VARCHAR(10),
    is_available BOOLEAN DEFAULT TRUE,
    is_ac BOOLEAN DEFAULT FALSE,
    FOREIGN KEY (room_id) REFERENCES rooms(room_id)
);

CREATE TABLE bookings (
    booking_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    bed_id INT,
    hostel_id INT,
    check_in DATE,
    check_out DATE,
    status ENUM('Confirmed','Pending','Cancelled','Completed'),
    FOREIGN KEY (user_id) REFERENCES users(user_id),
    FOREIGN KEY (bed_id) REFERENCES beds(bed_id),
    FOREIGN KEY (hostel_id) REFERENCES hostels(hostel_id)
);

CREATE TABLE payments (
    payment_id INT AUTO_INCREMENT PRIMARY KEY,
    booking_id INT,
    user_id INT,
    amount DECIMAL(10,2),
    method ENUM('UPI','Card','NetBanking','Cash'),
    status ENUM('Success','Failed','Pending'),
    payment_date DATE,
    FOREIGN KEY (booking_id) REFERENCES bookings(booking_id),
    FOREIGN KEY (user_id) REFERENCES users(user_id)
);

PART 5 - VIEWS

CREATE OR REPLACE VIEW view_hostel_overview AS
SELECT 
    h.name,
    h.city,
    r.room_type,
    r.rent_per_bed,
    GROUP_CONCAT(a.name) AS amenities
FROM hostels h
JOIN rooms r ON h.hostel_id = r.hostel_id
LEFT JOIN hostel_amenities ha ON h.hostel_id = ha.hostel_id
LEFT JOIN amenities a ON ha.amenity_id = a.amenity_id
GROUP BY h.hostel_id, r.room_id;

CREATE OR REPLACE VIEW view_booking_details AS
SELECT 
    b.booking_id,
    u.name AS user_name,
    h.name AS hostel_name,
    r.room_type
FROM bookings b
JOIN users u ON b.user_id = u.user_id
JOIN hostels h ON b.hostel_id = h.hostel_id
JOIN beds bd ON b.bed_id = bd.bed_id
JOIN rooms r ON bd.room_id = r.room_id;

CREATE OR REPLACE VIEW view_room_summary AS
SELECT 
    h.name,
    COUNT(r.room_id) AS rooms,
    AVG(r.rent_per_bed) AS avg_rent
FROM hostels h
LEFT JOIN rooms r ON h.hostel_id = r.hostel_id
GROUP BY h.hostel_id;

CREATE OR REPLACE VIEW view_payment_summary AS
SELECT 
    method,
    COUNT(*) AS total,
    SUM(amount) AS total_amount
FROM payments
GROUP BY method;

DELIMITER //

CREATE PROCEDURE GetHostelsByFilter(
    IN p_city VARCHAR(50),
    IN p_max_rent DECIMAL(8,2)
)
BEGIN
    SELECT 
        h.name,
        h.type,
        r.room_type,
        r.rent_per_bed,
        GROUP_CONCAT(DISTINCT a.name) AS amenities
    FROM hostels h
    JOIN rooms r ON h.hostel_id = r.hostel_id
    LEFT JOIN hostel_amenities ha ON h.hostel_id = ha.hostel_id
    LEFT JOIN amenities a ON ha.amenity_id = a.amenity_id
    WHERE h.city = p_city 
      AND r.rent_per_bed <= p_max_rent
    GROUP BY h.hostel_id, r.room_id;
END //

PART 7 - TRIGGERS

CREATE TABLE payment_audit (
    audit_id INT AUTO_INCREMENT PRIMARY KEY,
    payment_id INT,
    action_type VARCHAR(10),
    old_amount DECIMAL(10,2),
    new_amount DECIMAL(10,2),
    changed_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

DELIMITER //

CREATE TRIGGER trg_validate_payment_insert
BEFORE INSERT ON payments
FOR EACH ROW
BEGIN
    IF NEW.amount < 0 THEN
        SIGNAL SQLSTATE '45000' 
        SET MESSAGE_TEXT = 'Amount cannot be negative';
    END IF;
END //

CREATE TRIGGER trg_audit_payment_update
AFTER UPDATE ON payments
FOR EACH ROW
BEGIN
    IF OLD.amount <> NEW.amount THEN
        INSERT INTO payment_audit(
            payment_id,
            action_type,
            old_amount,
            new_amount
        )
        VALUES(
            NEW.payment_id,
            'UPDATE',
            OLD.amount,
            NEW.amount
        );
    END IF;
END //

CREATE TRIGGER trg_audit_payment_delete
AFTER DELETE ON payments
FOR EACH ROW
BEGIN
    INSERT INTO payment_audit(
        payment_id,
        action_type,
        old_amount,
        new_amount
    )
    VALUES(
        OLD.payment_id,
        'DELETE',
        OLD.amount,
        NULL
    );
END //

DELIMITER ;

 FROM bookings b

 bd ON b.bed_id=bd.bed_id JOIN rooms r
